# Pinned to exact versions so builds are reproducible. Previously every entry
# used >=, which meant two installs weeks apart could resolve different
# dependency trees and behave differently.
#
# Install:  pip install -r requirements.txt

# --- Web framework ---
# starlette is imported directly by two route modules but deliberately not
# listed: fastapi pins an exact starlette version, and a second pin here would
# either duplicate it or collide with it on the next fastapi bump. It is
# guaranteed installed as fastapi's own dependency.
fastapi==0.136.0
uvicorn[standard]==0.45.0
python-multipart==0.0.26

# --- Database ---
SQLAlchemy==2.0.49
alembic==1.18.4
asyncpg==0.31.0
aiosqlite==0.22.1

# --- Validation / settings ---
pydantic==2.13.3
pydantic-settings==2.14.0
# EmailStr in schemas/user.py and schemas/access_request.py needs this, and
# pydantic does not pull it in — it raises at import time instead:
#   ImportError: email-validator is not installed
# It was present locally as somebody else's transitive dependency, so nothing
# failed until a clean install on the deploy host.
email-validator==2.3.0

# --- Auth ---
python-jose[cryptography]==3.5.0
passlib[bcrypt]==1.7.4
# passlib 1.7.4 reads bcrypt.__about__, which bcrypt 4.1+ removed. Pinning to
# 4.0.1 avoids the resulting startup warning and hashing failure.
bcrypt==4.0.1

# --- Rate limiting (brute-force protection on login) ---
slowapi==0.1.10

# --- Image validation and sanitisation ---
